package chapter14;

import java.io.IOException;
import java.io.PrintWriter;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

import javax.naming.InitialContext;
import javax.sql.DataSource;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import tool.Page;

@WebServlet("/chapter14/practice01")
public class Practice01 extends HttpServlet {
	private static final long serialVersionUID = 1L;

	protected void doGet(
		HttpServletRequest request, HttpServletResponse response)
	throws ServletException, IOException {
		PrintWriter out = response.getWriter();
		Page.header(out);
		
		String sort=request.getParameter("sort");
		String sql="select * from product";
		
		if("high".equals(sort)) {
			sql += " order by price desc";
		} else if("low".equals(sort)) {
			sql += " order by price asc";
		}
		
		try {
			InitialContext ic = new InitialContext();
			DataSource ds = (DataSource) ic.lookup(
					"java:/comp/env/jdbc/book");
			Connection con = ds.getConnection();
			
			PreparedStatement st=con.prepareStatement(sql);
			ResultSet rs=st.executeQuery();
			
			while(rs.next()) {
				out.println(rs.getInt("id") + ":" + rs.getString("name") + ":" + rs.getInt("price") + "円<br>");
			}
			
			st.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace(out);
        }
        Page.footer(out);
	}

}

package chapter14;

import java.io.IOException;
import java.io.PrintWriter;
import java.sql.Connection;
import java.sql.PreparedStatement;

import javax.naming.InitialContext;
import javax.sql.DataSource;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import tool.Page;

@WebServlet("/chapter14/Select")
public class Select extends HttpServlet {
	private static final long serialVersionUID = 1L;

	protected void doGet(
		HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		
	}

	protected void doPost(
		HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		
		response.setContentType("text/html; charset=UTF-8");
		PrintWriter out=response.getWriter();
		
		request.setCharacterEncoding("UTF-8");
		String count=request.getParameter("count");
		String payment=request.getParameter("payment");
		String review=request.getParameter("review");
		String mailParam=request.getParameter("mail");
		
		String mail = (mailParam != null)? "希望する":"希望しない";
		
		Page.header(out);
		
		try {
			InitialContext ic = new InitialContext();
			DataSource ds = (DataSource) ic.lookup(
					"java:/comp/env/jdbc/book");
			Connection con = ds.getConnection();
			
			PreparedStatement st=con.prepareStatement(
					"insert into purchase(count, payment,review,mail) values(?,?,?,?)"
					);
			st.setInt(1, Integer.parseInt(count));
			st.setString(2, payment);
			st.setString(3,review);
			st.setString(4, mail);
			
			int line=st.executeUpdate();
			
			if(line>0){
				out.println("<h3>注文内容をデータベースに保存しました！</h3>");
				out.println("<p>購入数: " + count + " 個</p>");
				out.println("<p>お支払方法: " + payment + "</p>");
				out.println("<p>ご感想: 「" + review + "」</p>");
				out.println("<p>お知らせメール: " + mail + "</p>");
			} else {
				out.println("<p>データベースへの保存に失敗しました。</p>");
			}
			
			st.close();
			con.close();
			
		} catch (Exception e) {
			e.printStackTrace(out);
		}
		Page.footer(out);
	}

}

【応用問題１】				
				
概要				
シンプルな掲示板を作成する。				
ユーザーが名前と投稿内容を入力し、掲示板に投稿できる。				
投稿は一覧表示され、新しい投稿が上に表示される。				
データはデータベースに保存される。				
				
画面イメージ				
画面デザインについて、画像と同じでなくても問題ございません。				
見やすいレイアウトを心掛けてください。				

テーブル設計（データベース設計）				
	posts テーブル（投稿情報）			
	カラム名	データ型	制約	説明
	id	INT	PRIMARY KEY, AUTO_INCREMENT	投稿の一意識別子
	name	VARCHAR(50)	NOT NULL	投稿者名
	content	TEXT	NOT NULL	投稿内容
	post_date	TIMESTAMP	DEFAULT CURRENT_TIMESTAMP	投稿日時
				
 画面設計（JSP）				
	① 掲示板画面（index.jsp）			
		機能		
			・投稿フォーム	
			・投稿一覧表示	
		項目		
			・名前（テキストボックス）	
			・本文（テキストエリア）	
			・投稿ボタン	
			・投稿一覧（投稿者名、本文、投稿日時）	
				
必要な機能				
	① 投稿機能			
		フォームに入力されたデータをサーブレット経由でデータベースに保存する。		
		投稿後、掲示板画面へリダイレクトする。		
	② 投稿一覧表示			
		データベースに保存された投稿を取得し、一覧で表示する。		
		新しい投稿が上に来るように並べる。		
	③ 入力チェック（バリデーション）			
		名前と本文は空欄で投稿できないようにする。		
	④ 文字化け対策			
		日本語が正しく表示されるように文字エンコーディングをUTF-8に設定する。		
				
				
発展課題（余力があれば）				
	・削除機能（投稿を削除できるようにする）			
	・MVCモデルを使用する
